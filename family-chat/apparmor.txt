#include <tunables/global>

# --- Verification history ---
# Shipped in "complain" mode initially (2.29.0) — written from this
# app's actual Dockerfile/run.sh and known file/network footprint, but
# never run against a real AppArmor-enabled host before that release,
# so it logged instead of blocking anything it didn't explicitly cover,
# per Home Assistant's own documented process for a new custom profile.
#
# Verified clean on a real instance: profile confirmed loaded
# (Supervisor's own log showed "Adding/updating AppArmor profile:
# c2f85679_family_chat"), then exercised deliberately broad real usage
# — chat, file upload, avatar upload, the admin panel, adding a
# calendar event (outbound to Home Assistant's own Supervisor API),
# and posting a GIF (outbound to GIPHY's API) — covering every distinct
# code path this app has, not just the common chat one. `dmesg | grep
# -i apparmor` stayed empty throughout, which in complain mode
# specifically means nothing fell outside the rules below (complain
# mode logs exactly the actions that would've been denied — it doesn't
# log ordinary, already-permitted activity, so "nothing logged" here is
# a real positive result, not just an absence of testing). Switched to
# enforcing on that basis.
#
# If a future version of this app does something new — a different
# outbound API, a new file location, anything — and something breaks
# after that, that's the first place to look: add whatever's missing
# to the rules below based on what actually failed, the same way this
# profile was verified the first time, rather than guessing.
profile family_chat flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/python>

  # Inbound HTTP/WebSocket on the ingress port; outbound to Home
  # Assistant's own Supervisor API (calendar, notifications) and to
  # GIPHY's API when a key is configured. Nothing in this app ever
  # needs a raw socket.
  network inet stream,
  network inet6 stream,
  deny network raw,

  # The actual startup chain, matching this app's real Dockerfile —
  # plain python:3.11-alpine, not one of the s6-overlay-based add-on
  # base images, so there's no bashio/s6 supervision tree to account
  # for here the way most add-on profiles need.
  /bin/sh ix,
  /run.sh r,
  /usr/local/bin/python3* ix,
  /usr/local/lib/python3.11/** r,
  /usr/local/lib/python3.11/**.so* mr,
  /usr/lib/*.so* mr,
  /lib/*.so* mr,

  # The app's own code — read-only. This process never has a
  # legitimate reason to modify its own source after the image is
  # built.
  /app/** r,

  # Everything this app ever persists lives under /data and nowhere
  # else: the add-on's own configuration (read-only — this app never
  # writes it), the SQLite database plus its journal/WAL sidecar files,
  # uploaded files, and the generated session-signing secret.
  /data/options.json r,
  /data/chat.db rwk,
  /data/chat.db-journal rwk,
  /data/chat.db-wal rwk,
  /data/chat.db-shm rwk,
  /data/uploads/** rwk,
  /data/secret_key rw,

  # Explicit defense-in-depth denials — this app has no legitimate
  # reason to ever touch Home Assistant's own core config, another
  # add-on's TLS material, or write to /proc or /sys.
  deny /data/homeassistant rw,
  deny /data/ssl rw,
  deny /proc/** wl,
  deny /sys/** wl,

  / r,
  /etc/resolv.conf r,
  /tmp/** rwk,
}

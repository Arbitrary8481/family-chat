#include <tunables/global>

# --- Verification history ---
# Shipped in "complain" mode initially (2.29.0) — verified against a
# real instance across every distinct feature this app has, then
# switched to enforcing (2.29.1) on that basis.
#
# Reverted back to complain mode (2.29.2) as a precaution when the
# add-on failed to start right after that — turned out to be an
# unrelated bug in the python:3.11-alpine base image itself (see the
# Dockerfile's own comment, and CHANGELOG 2.29.2), not this profile;
# AppArmor denials produce "Permission denied", not the "No such file
# or directory" error that actually occurred. Reverted anyway at the
# time since there was no way to be fully certain from outside a real
# instance, and complain mode costs nothing to fall back to.
#
# Re-verified on the fixed build (2.29.3, pinned base image): same
# real-usage pass as the first verification — chat, file upload,
# avatar upload, the admin panel, adding a calendar event, posting a
# GIF — covering every distinct code path this app has. dmesg stayed
# clean of any AppArmor activity throughout, which in complain mode
# specifically means nothing fell outside the rules below. Switched
# back to enforcing on that basis.
profile family_chat flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/python>

  # Inbound HTTP/WebSocket on the ingress port; outbound to Home
  # Assistant's own Supervisor API (calendar, notifications) and to
  # GIPHY's API when a key is configured. Nothing in this app ever
  # needs a raw socket.
  network inet stream,
  network inet6 stream,
  deny network raw,

  # The actual startup chain, matching this app's real Dockerfile —
  # plain python:3.11-alpine, not one of the s6-overlay-based add-on
  # base images, so there's no bashio/s6 supervision tree to account
  # for here the way most add-on profiles need.
  /bin/sh ix,
  /run.sh r,
  /usr/local/bin/python3* ix,
  /usr/local/lib/python3.11/** r,
  /usr/local/lib/python3.11/**.so* mr,
  /usr/lib/*.so* mr,
  /lib/*.so* mr,

  # The app's own code — read-only. This process never has a
  # legitimate reason to modify its own source after the image is
  # built.
  /app/** r,

  # Everything this app ever persists lives under /data and nowhere
  # else: the add-on's own configuration (read-only — this app never
  # writes it), the SQLite database plus its journal/WAL sidecar files,
  # uploaded files, and the generated session-signing secret.
  /data/options.json r,
  /data/chat.db rwk,
  /data/chat.db-journal rwk,
  /data/chat.db-wal rwk,
  /data/chat.db-shm rwk,
  /data/uploads/** rwk,
  /data/secret_key rw,

  # Explicit defense-in-depth denials — this app has no legitimate
  # reason to ever touch Home Assistant's own core config, another
  # add-on's TLS material, or write to /proc or /sys.
  deny /data/homeassistant rw,
  deny /data/ssl rw,
  deny /proc/** wl,
  deny /sys/** wl,

  / r,
  /etc/resolv.conf r,
  /tmp/** rwk,
}

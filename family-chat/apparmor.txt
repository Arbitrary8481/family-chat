#include <tunables/global>

# --- Verification history ---
# Shipped in "complain" mode initially (2.29.0) — verified against a
# real instance, switched to enforcing (2.29.1) on that basis.
#
# That crashed the add-on outright: "Error loading shared library
# libpython3.11.so.1.0" / "Error relocating ...: Py_BytesMain: symbol
# not found". Wrongly attributed at the time (2.29.2) to an unrelated
# python:3.11-alpine base-image bug, based on the assumption that an
# AppArmor denial always produces "Permission denied" rather than "No
# such file or directory" — that assumption doesn't hold for a denied
# *executable mapping* of a library specifically (how the dynamic
# loader actually loads a shared library, as opposed to a plain file
# read), which can surface through the loader's own generic error
# message instead. The real evidence was the timing, not the base
# image: this crashed exactly when enforcing was turned on (2.29.1),
# went away when reverted to complain mode *together with* the
# (unnecessary, in hindsight) Dockerfile pin (2.29.2), and came right
# back the instant enforcing was turned on again (2.29.4) with that
# same Dockerfile pin still in place, unchanged. That's AppArmor, not
# the base image.
#
# The actual gap: libpython3.11.so.1.0 lives directly in
# /usr/local/lib/ on this image. The rules below only ever covered the
# /usr/local/lib/python3.11/ *subdirectory*, never /usr/local/lib/
# itself — so the interpreter's own shared library was never actually
# granted. Fixed at 2.29.6.
#
# Why complain mode hadn't already caught this in two earlier "clean"
# passes turned out to have a real explanation, discovered later and
# unrelated to this profile itself: Home Assistant Supervisor's own
# local git clone of this add-on's repository had fallen out of sync
# with GitHub, and app/static/script.js specifically had been silently
# stale across many versions — confirmed by inspecting the actual
# browser-loaded source, which was missing features known to exist in
# the real repo. It's plausible the deployed container's overall
# footprint during those earlier verification passes wasn't fully
# representative of the intended build for the same reason. Fixed by
# removing and re-adding the repository in the add-on store, forcing a
# fresh clone (confirmed via the same script.js inspection).
#
# Re-verified from that genuinely fresh, correctly-synced build
# (2.29.9): real usage across every feature this app has, specifically
# including a deliberate container restart (the exact scenario the
# missing-library-rule bug lived in) followed by the full feature pass.
# dmesg stayed clean of any AppArmor activity throughout. This is the
# first verification pass done with real confidence that the code
# under test actually matched what was intended. Switched to enforcing
# on that basis.
#
# That broke immediately anyway (2.29.11): "Fatal Python error:
# init_fs_encoding: failed to get the Python codec of the filesystem
# encoding" / "ModuleNotFoundError: No module named 'encodings'" — the
# interpreter can't even complete its own startup, before any
# application code runs. Reverted to complain mode again pending real
# evidence (a dmesg check from the actual failure) rather than guessing
# at the specific rule a third time.
profile family_chat flags=(attach_disconnected,mediate_deleted,complain) {
  #include <abstractions/base>
  #include <abstractions/python>

  # Inbound HTTP/WebSocket on the ingress port; outbound to Home
  # Assistant's own Supervisor API (calendar, notifications) and to
  # GIPHY's API when a key is configured. Nothing in this app ever
  # needs a raw socket.
  network inet stream,
  network inet6 stream,
  deny network raw,

  # The actual startup chain, matching this app's real Dockerfile —
  # plain python:3.11-alpine, not one of the s6-overlay-based add-on
  # base images, so there's no bashio/s6 supervision tree to account
  # for here the way most add-on profiles need.
  /bin/sh ix,
  /run.sh r,
  /usr/local/bin/python3* ix,
  # The interpreter's own shared library (libpython3.11.so.1.0) lives
  # directly here, as a sibling of the python3.11/ subdirectory below —
  # not inside it. This is the rule that was missing.
  /usr/local/lib/*.so* mr,
  /usr/local/lib/python3.11/** r,
  /usr/local/lib/python3.11/**.so* mr,
  /usr/lib/*.so* mr,
  /lib/*.so* mr,

  # The app's own code — read-only. This process never has a
  # legitimate reason to modify its own source after the image is
  # built.
  /app/** r,

  # Everything this app ever persists lives under /data and nowhere
  # else: the add-on's own configuration (read-only — this app never
  # writes it), the SQLite database plus its journal/WAL sidecar files,
  # uploaded files, and the generated session-signing secret.
  /data/options.json r,
  /data/chat.db rwk,
  /data/chat.db-journal rwk,
  /data/chat.db-wal rwk,
  /data/chat.db-shm rwk,
  /data/uploads/** rwk,
  /data/secret_key rw,

  # Explicit defense-in-depth denials — this app has no legitimate
  # reason to ever touch Home Assistant's own core config, another
  # add-on's TLS material, or write to /proc or /sys.
  deny /data/homeassistant rw,
  deny /data/ssl rw,
  deny /proc/** wl,
  deny /sys/** wl,

  / r,
  /etc/resolv.conf r,
  /tmp/** rwk,
}
